cmake_minimum_required(VERSION 3.16)
project(flighting_rasterix_bridge C)
set(FLIGHTING_PLATFORM_SOURCE "" CACHE FILEPATH "GPU-team source implementing fg_platform_* hooks")
set(FLIGHTING_PLATFORM_LIBRARIES "" CACHE STRING "RasterIX/host libraries for the selected revision")
if(NOT EXISTS "${FLIGHTING_PLATFORM_SOURCE}")
  message(FATAL_ERROR "Provide FLIGHTING_PLATFORM_SOURCE for the selected RasterIX software/board context. No fake host is supplied.")
endif()
add_library(flighting_rasterix_bridge SHARED gl_bridge.c "${FLIGHTING_PLATFORM_SOURCE}")
target_include_directories(flighting_rasterix_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(flighting_rasterix_bridge PRIVATE ${FLIGHTING_PLATFORM_LIBRARIES})
set_target_properties(flighting_rasterix_bridge PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
